# Generic shapes for the AutoOS publication scanner. TRACKED: safe to commit.
# One name<TAB>regex per line. Names describe the kind of hit and never echo the value.
# Private literals (usernames, emails, domains, IPs, paths) are in patterns.private.txt (gitignored).

rfc1918	\b(10\.\d{1,3}|192\.168|172\.(1[6-9]|2\d|3[01]))\.\d{1,3}\.\d{1,3}\b
win-home	(?i)[a-z]:[\/]+Users[\/]+(?!(you|x|me|someone|user|username|<|\.\.\.|%|\$))[^\/\s\"'`<>|]+
unix-home	/home/(?!(you|someone|user|username)\b)[a-z][a-z0-9_-]+
key-shape	(sk-or-v1-[0-9a-f]{16,}|sk-ant-[A-Za-z0-9-]{16,}|LLM_[A-Za-z0-9]{12,})
